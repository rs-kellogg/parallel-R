LINUX COMMANDS
----------------
# find present working directory
pwd

# ---- run R on KLC ----

# check modules loaded
module list 

# search for a module (scroll down until you find your software; typing "G" will end the command
module avail R

# load a module
module load R/4.2.0

# launch Rstudio
rstudio

# launch an Rscript
Rscript sample.R

# --- get class materials ----

# clone materials from git repo
module load git/2.37.2
git clone https://github.com/rs-kellogg/parallel-R

# navigate into directory and cpu subfolder
cd parallel-R/cpu

# ---- create an isolated conda env for your project ----

# install a package manager
module load mamba

# use a yaml or recipe file to install all packages at once
mamba env create -f r_parallel.yml


# ---- activate an env ----
# activate the env with mamba or conda
mamba init 
# OR
conda init

# activate the conda env without running mamba/conda init today
eval "$('/hpc/software/mamba/24.3.0/bin/conda' 'shell.bash' 'hook' 2> /dev/null)"
source "/hpc/software/mamba/24.3.0/etc/profile.d/mamba.sh"

# activate your env
mamba activate r_parallel 
# OR
conda activate r_parallel

# activate the env Research Support shared with you
conda activate /kellogg/software/envs/r_parallel
# OR
mamba activate /kellogg/software/envs/r_parallel

# exit a conda env
conda deactivate
# OR
mamba deactivate

# ---- monitor your code ----

# check processes running on a node 
top
top -u <your_netID> -i -c # use this for a more precise view of just your own processes
(Control + c to exit out of this command)

# nano text editor
nano sample.R
(save changes: Control + x, Yes, Enter)
