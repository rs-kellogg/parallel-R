------------------------
SETTING UP THE CONDA ENV
------------------------
# Since the software and packages in this environment require dependencies to be installed before running, we'll forego the .yml file

# ---- Initial Setup Only ----

# initial setup steps 
module load mamba
mamba create -n r_gpu_env r-base=4.4.0 r-rstudio-desktop -c conda-forge -y
mamba activate r_gpu_env

# load the compatible CUDA module
module load cuda/11.8.0-gcc-12.3.0 # you need to load a compatible CUDA module for the node

# install R packages
Rscript -e "install.packages('remotes', repos='https://cloud.r-project.org')"
Rscript -e "remotes::install_version('torch', version = '0.12.0', repos='https://cloud.r-project.org')"
Rscript -e "install.packages('tictoc', repos='https://cloud.r-project.org')"

# pre-download the Libtorch binaries
Rscript -e "torch::install_torch()"

# ---- Future Runs ----

module load mamba

mamba activate r_gpu_env OR
mamba activate /kellogg/software/envs/r_gpu_env # for our shared env

module load cuda/11.8.0-gcc-12.3.0
